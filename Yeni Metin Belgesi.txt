    function getLastThursdayOfMonth(year, month) {
      // month: 0-11
      const lastDay = new Date(year, month + 1, 0); // Ayın son günü
      let day = lastDay.getDay(); // 0: Pazar, 1: Pzt, ..., 4: Perşembe
      let diff = (day >= 4) ? (day - 4) : (day + 3);
      return new Date(year, month, lastDay.getDate() - diff, 23, 59, 59);
    }

    function checkDeadlineInfo() {
      const now = new Date();
      const currentYear = now.getFullYear();
      const currentMonth = now.getMonth();
      const lastThursday = getLastThursdayOfMonth(currentYear, currentMonth);

      const monthsTr = ['Ocak', 'Şubat', 'Mart', 'Nisan', 'Mayıs', 'Haziran', 'Temmuz', 'Ağustos', 'Eylül', 'Ekim', 'Kasım', 'Aralık'];
      const deadlineStr = `${lastThursday.getDate()} ${monthsTr[currentMonth]} ${currentYear} Perşembe (Saat 23:59)`;
      
      const isPast = now > lastThursday;
      const el = document.getElementById('deadlineDisplay');
      if (el) {
        el.innerHTML = `Bu Ayki Son Rapor Tarihi: <strong>${deadlineStr}</strong> ${isPast ? '<span style="color:#f87171;">(Süre Doldu)</span>' : '<span style="color:#4ade80;">(Girişler Açık)</span>'}`;
      }
      return { lastThursday, isPast, monthIndex: currentMonth, year: currentYear };
    }

       ADMİN İŞLEMLERİ: OKUL & FAALİYET TANIMLAMA
       ========================================================= */
    window.saveSchool = function(e) {
      e.preventDefault();
      const code = document.getElementById('sch_code').value.trim();
      const name = document.getElementById('sch_name').value.trim();
      const level = document.getElementById('sch_level').value;
      const pass = document.getElementById('sch_pass').value.trim();
      const teacher = document.getElementById('sch_teacher').value.trim();

      const existingIndex = window.db.schools.findIndex(s => s.code === code);
      if (existingIndex >= 0) {
        window.db.schools[existingIndex] = { code, name, level, pass, teacher };
        notify('Okul bilgileri güncellendi.');
      } else {
        window.db.schools.push({ code, name, level, pass, teacher });
        notify('Yeni okul ve koordinatör kaydedildi.');
      }
      saveAll();
      e.target.reset();
    };

    window.saveBulkSchools = function() {
      const text = document.getElementById('bulk_school_input').value.trim();
      if (!text) { alert('Lütfen metin giriniz.'); return; }
      const lines = text.split('\n');
      let count = 0;
      lines.forEach(line => {
        const parts = line.split(',').map(s => s.trim());
        if (parts.length >= 5) {
          const [code, name, level, teacher, pass] = parts;
          const idx = window.db.schools.findIndex(s => s.code === code);
          if (idx >= 0) {
            window.db.schools[idx] = { code, name, level, teacher, pass };
          } else {
            window.db.schools.push({ code, name, level, teacher, pass });
          }
          count++;
        }
      });
      saveAll();
      document.getElementById('bulk_school_input').value = '';
      notify(`${count} adet okul başarıyla tanımlandı.`);
    };

    window.restoreBackupJSON = function(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(evt) {
        try {
          const parsed = JSON.parse(evt.target.result);
          if (parsed.schools && parsed.activities) {
            window.db.schools = parsed.schools;
            window.db.activities = parsed.activities;
            if (parsed.settings) window.db.settings = parsed.settings;
            if (parsed.templates) window.db.activityTemplates = parsed.templates;
            if (parsed.documents) window.db.documents = parsed.documents;
            saveAll();
            notify('Veritabanı yedekten başarıyla geri yüklendi!');
          }
        } catch(err) {
          alert('Hata: ' + err.message);
        }
      };
      reader.readAsText(file);
    };

              <td>${s.name}</td>
